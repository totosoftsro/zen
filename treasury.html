Účet na míru

Vytvořte uživateli účet jedním voláním

create_account.sh
# Otevření účtu s IBAN pro uživatele platformy
curl https://api.zen.cz/v1/treasury/financial_accounts \
  -u "sk_live_••••••••:" \
  -d supported_currencies[]="czk" \
  -d supported_currencies[]="eur" \
  -d features[card_issuing][requested]=true \
  -d features[inbound_transfers][ach][requested]=true \
  -d metadata[user_id]="usr_8821"
