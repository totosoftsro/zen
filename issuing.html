create_card.sh
# Vytvoření virtuální karty se spend controls
curl https://api.zen.cz/v1/issuing/cards \
  -u "sk_live_••••••••:" \
  -d cardholder="ich_1Nov4k" \
  -d currency="czk" \
  -d type="virtual" \
  -d spending_controls[spending_limits][0][amount]=5000000 \
  -d spending_controls[spending_limits][0][interval]="monthly" \
  -d spending_controls[allowed_categories][]="transport"
